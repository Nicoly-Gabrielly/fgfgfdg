import random import time def mostrar_cabecalho(): print("=" * 40) print(" RPG DE ADIVINHAÇÃO - O LABIRINTO") print("=" * 40) def jogar_rpg(): mostrar_cabecalho() nome_jogador = input("\nDigite o nome do seu herói: ").strip() if not nome_jogador: nome_jogador = "Guerreiro Anônimo" vida_jogador = 100 vida_monstro = 50 pocoes = 2 print(f"\nSeja bem-vindo, {nome_jogador}!") print("Um monstro sombrio bloqueia o seu caminho. Vença-o para escapar!\n") time.sleep(1) while vida_jogador > 0 and vida_monstro > 0: print("-" * 40) print(f"Sua Vida: {vida_jogador} HP | Poções: {pocoes}") print(f"Vida do Monstro: {vida_monstro} HP") print("-" * 40) print("\nEscolha sua ação:") print("1. Atacar (Adivinhar o número mágico de 1 a 5)") print("2. Usar Poção (+25 de Vida)") escolha = input("\nO que deseja fazer? (1 ou 2): ").strip() if escolha == "1": try: palpite = int(input("Escolha um número entre 1 e 5: ")) except ValueError: print("⚠️ Por favor, digite apenas números inteiros!") continue if palpite < 1 or palpite > 5: print("⚠️ Número fora do alcance! Escolha de 1 a 5.") continue numero_secreto = random.randint(1, 5) print(f"\nO número secreto era: {numero_secreto}") if palpite == numero_secreto: dano = random.randint(15, 25) vida_monstro -= dano print(f"🎯 ATAQUE CRÍTICO! Você acertou o palpite e causou {dano} de dano!") else: dano_recebido = random.randint(10, 20) vida_jogador -= dano_recebido print(f"❌ Você errou! O monstro contra-atacou e você perdeu {dano_recebido} de vida.") elif escolha == "2": if pocoes > 0: cura = 25 vida_jogador = min(100, vida_jogador + cura) pocoes -= 1 print(f"\n🧪 Você usou uma poção e recuperou {cura} de HP! (Poções restantes: {pocoes})") else: print("\n⚠️ Você não tem mais poções!") else: print("\n⚠️ Opção inválida! Escolha 1 para atacar ou 2 para poção.") time.sleep(1.5) # Fim de jogo print("\n" + "=" * 40) if vida_jogador > 0: print(f"🏆 PARABÉNS, {nome_jogador.upper()}! Você derrotou o monstro!") else: print("☠️ GAME OVER! O monstro te venceu desta vez.") print("=" * 40) if __name__ == "__main__": jogar_rpg()
